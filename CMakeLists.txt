cmake_minimum_required(VERSION 3.20.0)

# --- FIX: Prevent ExecuTorch from exporting targets and crashing Zephyr ---
set(EXECUTORCH_ENABLE_INSTALL OFF CACHE INTERNAL "")
set(CMAKE_SKIP_INSTALL_RULES TRUE)
# --------------------------------------------------------------------------

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(kws_pico2)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 1. Update sources to match the new nested folder structure
target_sources(app PRIVATE 
    src/main.c
    src/dcnn/model_runner.cpp
    src/dcnn/model.c
)

# 2. Add the dcnn directory to the include path
target_include_directories(app PRIVATE src/dcnn)

# 3. Link the core ExecuTorch library generated by the module
target_link_libraries(app PRIVATE executorch portable_ops_lib)
